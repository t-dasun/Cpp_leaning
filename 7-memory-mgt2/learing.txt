Circular Reference issue 
class B;

class A {
public:
    std::shared_ptr<B> b;
};

class B {
public:
    std::shared_ptr<A> a;
};
if both a,b shared pointer it couse issue Both keep each other alive forever. 
solution
class B;

class A {
public:
    std::shared_ptr<B> b;
};

class B {
public:
    std::weak_ptr<A> a;
};
// keep one as weak pointer so it keep observing other shared pointers controller
weak_ptr CANNOT directly access object using: *wp , wp->x it use lock() 
std::shared_ptr<int> sp = std::make_shared<int>(10);
std::weak_ptr<int> wp = sp;
auto temp = wp.lock(); //If object still exists: lock() -> shared_ptr else nullptr
if(temp) {
    std::cout << *temp;
}
//Observer pattern, Break circular references. Cache systems use this
