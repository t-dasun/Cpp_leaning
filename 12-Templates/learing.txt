Explicit Specialization
template <typename T>
T Max(T x, T y)
{
    return x > y ? x : y;
}
// Specialization declaration in header
template <> const char* Max<const char*>(const char* x, const char* y);
/ Specialization definition in cpp
template <> const char* Max<const char*>(const char* x, const char* y)
{
    return std::strcmp(x, y) > 0 ? x : y;
    // why we have explicit specialization becouse char pointer is pass so it compaer adress not value
    // strcmp use to make sure values are compare
}
